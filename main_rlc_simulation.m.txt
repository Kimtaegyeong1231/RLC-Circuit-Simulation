%% 1. 초기화 및 매개변수 설정
clear; clc; close all;

% 회로 소자 값 설정 (필요에 따라 이 값을 수정해보세요)
R = 20;       % 저항 (Ohm)
L = 0.1;      % 인덕터 (H)
C = 100e-6;   % 커패시터 (F) - 100uF
V0 = 10;      % 직류 전원 전압 (V)

%% 2. 전기공학 물리량 계산
% 감쇠 계수(Alpha) 및 고유 진동수(Omega_0)
alpha = R / (2 * L);
omega0 = 1 / sqrt(L * C);

% 시간 축 설정 (0초부터 0.05초까지 1,000개 구간으로 분할)
t = linspace(0, 0.05, 1000);

%% 3. 감쇠 조건 판단 및 커패시터 전압 Vc(t) 계산
if alpha > omega0
    % [과감쇠 (Overdamped)]
    state_str = '과감쇠 (Overdamped)';
    s1 = -alpha + sqrt(alpha^2 - omega0^2);
    s2 = -alpha - sqrt(alpha^2 - omega0^2);
    
    % 초기조건 Vc(0)=0, i(0)=0 반영한 해
    A1 = V0 * s2 / (s1 - s2);
    A2 = -V0 * s1 / (s1 - s2);
    Vc = V0 + A1*exp(s1*t) + A2*exp(s2*t);

elseif abs(alpha - omega0) < 1e-6
    % [임계감쇠 (Critically Damped)]
    state_str = '임계감쇠 (Critically Damped)';
    
    Vc = V0 - V0*(1 + alpha*t).*exp(-alpha*t);

else
    % [부족감쇠 (Underdamped)]
    state_str = '부족감쇠 (Underdamped)';
    wd = sqrt(omega0^2 - alpha^2); % 감쇠 고유 진동수
    
    Vc = V0 - V0*exp(-alpha*t).*(cos(wd*t) + (alpha/wd)*sin(wd*t));
end

%% 4. 결과 출력 및 시각화
% 콘솔에 상태 출력
fprintf('=== RLC 회로 분석 결과 ===\n');
fprintf('Alpha (감쇠 계수): %.2f rad/s\n', alpha);
fprintf('Omega_0 (고유 진동수): %.2f rad/s\n', omega0);
fprintf('회로 상태: %s\n', state_str);

% 그래프 생성
figure('Color', 'white');
plot(t * 1000, Vc, 'b-', 'LineWidth', 2); % 시간을 ms 단위로 변환
grid on;
hold on;

% 목표 전압(V0) 점선 표시
yline(V0, 'r--', '목표 전압 (10V)', 'LineWidth', 1.5);

% 그래프 라벨 및 제목 설정
xlabel('시간 (ms)', 'FontSize', 12);
ylabel('커패시터 전압 (V)', 'FontSize', 12);
title(['RLC 직렬회로 과도 응답 : ', state_str], 'FontSize', 14);
legend('커패시터 전압 V_c(t)', '입력 전압 V_0', 'Location', 'southeast');
